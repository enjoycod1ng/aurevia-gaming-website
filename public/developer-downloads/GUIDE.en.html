// Node.js 20+ server-side helper. Never bundle this file with credentials into a browser.
import {createHmac, timingSafeEqual} from 'node:crypto';

export function verifyCallback(secret, rawBody, headers, operation, operatorId, now = Date.now() / 1000) {
  if (!secret || !operatorId || !['balance','debit','credit','rollback'].includes(operation)) throw new Error('Wallet verifier is not configured');
  const timestamp = headers['x-pp-timestamp'];
  const signature = headers['x-pp-signature'];
  if (!/^\d{1,12}$/.test(timestamp ?? '') || Math.abs(now - Number(timestamp)) > 300 || !/^[a-f0-9]{64}$/.test(signature ?? '')) {
    throw new Error('Invalid callback signature or timestamp');
  }
  const raw = Buffer.isBuffer(rawBody) ? rawBody : Buffer.from(rawBody, 'utf8');
  const expected = createHmac('sha256', secret).update(timestamp + '.').update(raw).digest();
  if (!timingSafeEqual(expected, Buffer.from(signature, 'hex'))) throw new Error('Invalid callback signature');
  const payload = JSON.parse(raw.toString('utf8'));
  if (typeof payload?.request_id !== 'string' || !payload.request_id || payload.request_id !== headers['x-pp-request-id'] || payload.operator_id !== operatorId || payload.transaction_type !== operation) {
    throw new Error('Callback identity mismatch');
  }
  return payload;
}

export async function providerRequest(baseUrl, apiKey, method, path, body, idempotencyKey) {
  const base = new URL(baseUrl.endsWith('/') ? baseUrl : baseUrl + '/');
  if (base.protocol !== 'https:') throw new Error('Provider URL must use HTTPS');
  if (!path.startsWith('api/provider/v1/')) throw new Error('Invalid provider path');
  const headers = {'X-Provider-API-Key': apiKey, 'Accept': 'application/json'};
  if (body !== undefined) headers['Content-Type'] = 'application/json';
  if (idempotencyKey) headers['Idempotency-Key'] = idempotencyKey;
  const response = await fetch(new URL(path, base), {
    method, headers, body: body === undefined ? undefined : JSON.stringify(body),
    redirect: 'error', signal: AbortSignal.timeout(15000),
  });
  const result = await response.json();
  if (!response.ok) {
    const error = new Error(result.message ?? 'Provider request failed');
    error.status = response.status;
    error.code = result.error;
    error.retryAfter = response.headers.get('retry-after');
    throw error;
  }
  return result.data;
}

export function launchGame(baseUrl, apiKey, authenticatedPlayerId, returnUrl, idempotencyKey, symbol = 'wildframes', device = 'desktop') {
  if (!idempotencyKey) throw new Error('Persist one idempotency key per launch attempt and reuse it on retries');
  if (!['desktop','mobile'].includes(device)) throw new Error('Device must be desktop or mobile');
  return providerRequest(baseUrl, apiKey, 'POST', 'api/provider/v1/launch-url', {
    symbol, external_player_id: authenticatedPlayerId, currency: 'EUR', lang: 'en', device, return_url: returnUrl,
  }, idempotencyKey);
}

<?php
declare(strict_types=1);
// PHP 8+ with cURL. Server-side only; load credentials from your secret store.

function verifyCallback(string $secret, string $rawBody, array $headers, string $operation, string $operatorId, ?int $now = null): array {
    if ($secret === '' || $operatorId === '' || !in_array($operation, ['balance','debit','credit','rollback'], true)) throw new RuntimeException('Wallet verifier is not configured');
    $headers = array_change_key_case($headers, CASE_LOWER);
    $timestamp = $headers['x-pp-timestamp'] ?? '';
    $signature = $headers['x-pp-signature'] ?? '';
    if (!preg_match('/^\d{1,12}$/D', $timestamp) || abs(($now ?? time()) - (int)$timestamp) > 300 || !preg_match('/^[a-f0-9]{64}$/D', $signature)) {
        throw new RuntimeException('Invalid callback signature or timestamp');
    }
    if (!hash_equals(hash_hmac('sha256', $timestamp . '.' . $rawBody, $secret), $signature)) {
        throw new RuntimeException('Invalid callback signature');
    }
    $payload = json_decode($rawBody, true, 512, JSON_THROW_ON_ERROR);
    if (!is_string($payload['request_id'] ?? null) || $payload['request_id'] === '' || ($payload['request_id'] ?? null) !== ($headers['x-pp-request-id'] ?? null)
        || ($payload['operator_id'] ?? null) !== $operatorId || ($payload['transaction_type'] ?? null) !== $operation) {
        throw new RuntimeException('Callback identity mismatch');
    }
    return $payload;
}

function providerRequest(string $baseUrl, string $apiKey, string $method, string $path, ?array $body = null, ?string $idempotencyKey = null): array {
    if (parse_url($baseUrl, PHP_URL_SCHEME) !== 'https' || !str_starts_with($path, 'api/provider/v1/')) {
        throw new InvalidArgumentException('Invalid provider HTTPS URL or API path');
    }
    if (strpbrk($apiKey . ($idempotencyKey ?? ''), "\r\n") !== false) throw new InvalidArgumentException('Invalid header');
    $headers = ['X-Provider-API-Key: ' . $apiKey, 'Accept: application/json'];
    if ($body !== null) $headers[] = 'Content-Type: application/json';
    if ($idempotencyKey !== null) $headers[] = 'Idempotency-Key: ' . $idempotencyKey;
    $handle = curl_init(rtrim($baseUrl, '/') . '/' . $path);
    curl_setopt_array($handle, [CURLOPT_RETURNTRANSFER => true, CURLOPT_CUSTOMREQUEST => $method,
        CURLOPT_HTTPHEADER => $headers, CURLOPT_CONNECTTIMEOUT => 5, CURLOPT_TIMEOUT => 15,
        CURLOPT_FOLLOWLOCATION => false, CURLOPT_SSL_VERIFYPEER => true, CURLOPT_SSL_VERIFYHOST => 2]);
    if ($body !== null) curl_setopt($handle, CURLOPT_POSTFIELDS, json_encode($body, JSON_THROW_ON_ERROR));
    $raw = curl_exec($handle);
    $status = curl_getinfo($handle, CURLINFO_RESPONSE_CODE);
    $failed = $raw === false;
    $errorNumber = curl_errno($handle);
    curl_close($handle);
    if ($failed) throw new RuntimeException('Provider connection failed (cURL ' . $errorNumber . '); retry launch with the same idempotency key');
    $result = json_decode($raw, true, 512, JSON_THROW_ON_ERROR);
    if ($status < 200 || $status >= 300) throw new RuntimeException($result['message'] ?? 'Provider request failed', $status);
    return $result['data'];
}

function launchGame(string $baseUrl, string $apiKey, string $authenticatedPlayerId, string $returnUrl, string $idempotencyKey, string $symbol = 'wildframes', string $device = 'desktop'): array {
    if ($idempotencyKey === '') throw new InvalidArgumentException('Persist an idempotency key before launching');
    if (!in_array($device, ['desktop','mobile'], true)) throw new InvalidArgumentException('Device must be desktop or mobile');
    return providerRequest($baseUrl, $apiKey, 'POST', 'api/provider/v1/launch-url', [
        'symbol' => $symbol, 'external_player_id' => $authenticatedPlayerId, 'currency' => 'EUR',
        'lang' => 'en', 'device' => $device, 'return_url' => $returnUrl,
    ], $idempotencyKey);
}

// .NET 8+. Server-side only. Reuse this client; obtain credentials from a secret store.
using System.Net;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Text.RegularExpressions;

public sealed class AureviaProvider : IDisposable
{
    private readonly HttpClient http;
    private readonly Uri baseUri;
    private readonly string apiKey;

    public AureviaProvider(string baseUrl, string apiKey)
    {
        baseUri = new Uri(baseUrl.TrimEnd('/') + "/");
        if (baseUri.Scheme != "https" || baseUri.UserInfo != "" || baseUri.Query != "" || baseUri.Fragment != "")
            throw new ArgumentException("Use the assigned HTTPS provider base URL.");
        if (string.IsNullOrEmpty(apiKey) || apiKey.Contains('\r') || apiKey.Contains('\n'))
            throw new ArgumentException("Invalid API key header.");
        this.apiKey = apiKey;
        http = new HttpClient(new HttpClientHandler { AllowAutoRedirect = false, UseCookies = false })
            { Timeout = TimeSpan.FromSeconds(15) };
    }

    public async Task<JsonElement> RequestAsync(HttpMethod method, string path, object? body = null,
        string? idempotencyKey = null, CancellationToken cancellationToken = default)
    {
        if (!path.StartsWith("api/provider/v1/", StringComparison.Ordinal) || path.Contains("..") || path.Contains('\\'))
            throw new ArgumentException("Invalid provider path.");
        using var request = new HttpRequestMessage(method, new Uri(baseUri, path));
        request.Headers.Add("X-Provider-API-Key", apiKey);
        request.Headers.Add("Accept", "application/json");
        if (idempotencyKey != null) request.Headers.Add("Idempotency-Key", idempotencyKey);
        if (body != null) request.Content = new StringContent(JsonSerializer.Serialize(body), Encoding.UTF8, "application/json");
        using var response = await http.SendAsync(request, cancellationToken);
        if (!response.IsSuccessStatusCode)
            throw new ProviderException(response.StatusCode, response.Headers.RetryAfter?.ToString());
        using var document = JsonDocument.Parse(await response.Content.ReadAsStringAsync(cancellationToken));
        return document.RootElement.GetProperty("data").Clone();
    }

    public Task<JsonElement> LaunchAsync(string authenticatedPlayerId, string returnUrl, string persistedKey,
        string symbol = "wildframes", string device = "desktop", CancellationToken cancellationToken = default)
    {
        if (string.IsNullOrEmpty(persistedKey)) throw new ArgumentException("Persist a key before launching; reuse it with the exact body on retries.");
        if (device != "desktop" && device != "mobile") throw new ArgumentException("Invalid device.");
        return RequestAsync(HttpMethod.Post, "api/provider/v1/launch-url", new {
            symbol, external_player_id = authenticatedPlayerId, currency = "EUR", lang = "en", device, return_url = returnUrl
        }, persistedKey, cancellationToken);
    }

    public static JsonElement VerifyCallback(string secret, byte[] rawBody, string timestamp, string signature,
        string requestId, string operation, string operatorId, long? now = null)
    {
        if (string.IsNullOrEmpty(secret) || string.IsNullOrEmpty(operatorId) ||
            !new[] { "balance", "debit", "credit", "rollback" }.Contains(operation))
            throw new ArgumentException("Wallet verifier is not configured.");
        if (!Regex.IsMatch(timestamp, @"\A[0-9]{1,12}\z") ||
            Math.Abs((now ?? DateTimeOffset.UtcNow.ToUnixTimeSeconds()) - long.Parse(timestamp)) > 300 ||
            !Regex.IsMatch(signature, @"\A[a-f0-9]{64}\z")) throw new CryptographicException("Invalid signature or timestamp.");
        using var hmac = IncrementalHash.CreateHMAC(HashAlgorithmName.SHA256, Encoding.UTF8.GetBytes(secret));
        hmac.AppendData(Encoding.ASCII.GetBytes(timestamp + "."));
        hmac.AppendData(rawBody);
        if (!CryptographicOperations.FixedTimeEquals(hmac.GetHashAndReset(), Convert.FromHexString(signature)))
            throw new CryptographicException("Invalid signature.");
        using var document = JsonDocument.Parse(rawBody);
        var payload = document.RootElement;
        if (string.IsNullOrEmpty(requestId) || payload.GetProperty("request_id").GetString() != requestId ||
            payload.GetProperty("operator_id").GetString() != operatorId || payload.GetProperty("transaction_type").GetString() != operation)
            throw new InvalidOperationException("Callback identity mismatch.");
        return payload.Clone();
    }

    public void Dispose() => http.Dispose();
}

public sealed class ProviderException(HttpStatusCode status, string? retryAfter) : Exception($"Provider HTTP {(int)status}")
{
    public HttpStatusCode Status { get; } = status;
    public string? RetryAfter { get; } = retryAfter;
}

// Java 17+ and Jackson (see pom.xml). Backend only; never expose API keys to a browser.
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.net.URI;
import java.net.http.*;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.Duration;
import java.time.Instant;
import java.util.*;
import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;

public final class AureviaProvider {
    private static final ObjectMapper JSON = new ObjectMapper();
    private final HttpClient http = HttpClient.newBuilder().followRedirects(HttpClient.Redirect.NEVER)
        .connectTimeout(Duration.ofSeconds(5)).build();
    private final URI base;
    private final String key;

    public AureviaProvider(String baseUrl, String apiKey) {
        base = URI.create(baseUrl.replaceAll("/+$", "") + "/");
        if (!"https".equals(base.getScheme()) || base.getHost() == null || base.getUserInfo() != null ||
            base.getQuery() != null || base.getFragment() != null) throw new IllegalArgumentException("Use the assigned HTTPS base URL.");
        if (apiKey == null || apiKey.isEmpty() || apiKey.contains("\r") || apiKey.contains("\n"))
            throw new IllegalArgumentException("Invalid API key header.");
        key = apiKey;
    }

    public JsonNode request(String method, String path, Object body, String persistedKey) throws Exception {
        if (!path.startsWith("api/provider/v1/") || path.contains("..") || path.contains("\\"))
            throw new IllegalArgumentException("Invalid provider path.");
        var builder = HttpRequest.newBuilder(base.resolve(path)).timeout(Duration.ofSeconds(15))
            .header("X-Provider-API-Key", key).header("Accept", "application/json");
        if (persistedKey != null) builder.header("Idempotency-Key", persistedKey);
        var publisher = HttpRequest.BodyPublishers.noBody();
        if (body != null) {
            builder.header("Content-Type", "application/json");
            publisher = HttpRequest.BodyPublishers.ofString(JSON.writeValueAsString(body), StandardCharsets.UTF_8);
        }
        var response = http.send(builder.method(method, publisher).build(), HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
        if (response.statusCode() < 200 || response.statusCode() >= 300)
            throw new ProviderException(response.statusCode(), response.headers().firstValue("Retry-After").orElse(null));
        return JSON.readTree(response.body()).required("data");
    }

    public JsonNode launch(String authenticatedPlayerId, String returnUrl, String persistedKey, String symbol, String device) throws Exception {
        if (persistedKey == null || persistedKey.isEmpty()) throw new IllegalArgumentException("Persist the launch key before sending; reuse it on retries.");
        if (!Set.of("desktop", "mobile").contains(device)) throw new IllegalArgumentException("Invalid device.");
        return request("POST", "api/provider/v1/launch-url", Map.of("symbol", symbol,
            "external_player_id", authenticatedPlayerId, "currency", "EUR", "lang", "en", "device", device,
            "return_url", returnUrl), persistedKey);
    }

    public static JsonNode verifyCallback(String secret, byte[] rawBody, String timestamp, String signature,
        String requestId, String operation, String operatorId, long now) throws Exception {
        if (secret == null || secret.isEmpty() || operatorId == null || operatorId.isEmpty() ||
            !Set.of("balance", "debit", "credit", "rollback").contains(operation)) throw new IllegalArgumentException("Verifier is not configured.");
        if (timestamp == null || !timestamp.matches("[0-9]{1,12}") ||
            Math.abs(now - Long.parseLong(timestamp)) > 300 || signature == null || !signature.matches("[a-f0-9]{64}"))
            throw new SecurityException("Invalid signature or timestamp.");
        var mac = Mac.getInstance("HmacSHA256");
        mac.init(new SecretKeySpec(secret.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
        mac.update((timestamp + ".").getBytes(StandardCharsets.US_ASCII));
        if (!MessageDigest.isEqual(mac.doFinal(rawBody), HexFormat.of().parseHex(signature))) throw new SecurityException("Invalid signature.");
        var payload = JSON.readTree(rawBody);
        if (requestId == null || requestId.isEmpty() || !payload.path("request_id").isTextual() ||
            !requestId.equals(payload.path("request_id").asText()) || !operatorId.equals(payload.path("operator_id").asText()) ||
            !operation.equals(payload.path("transaction_type").asText())) throw new SecurityException("Callback identity mismatch.");
        return payload;
    }

    public static final class ProviderException extends Exception {
        public final int status;
        public final String retryAfter;
        ProviderException(int status, String retryAfter) {
            super("Provider HTTP " + status); this.status = status; this.retryAfter = retryAfter;
        }
    }
}

1. Understand the integration

Your backend authenticates the player, lists assigned games and requests a launch URL. Your frontend opens that URL. Your wallet remains the source of truth and implements four HTTPS callbacks: balance, debit, credit and rollback. The hosted game handles its own assets and native game protocol.

Release 1.0.1 is a sandbox with nine games and two simulated operator wallets. Use simulated funds only. It currently supports EUR and English game content, with four concurrent sessions overall and two per operator. Production client access requires separate acceptance and enablement. Documentation in Spanish or Portuguese does not enable those languages in the game API. This service wraps the VPS game runtime; it is not the official Play’n GO developer API.

2. Configure your backend

Use https://aureviagaming.com/playngo-sandbox as PROVIDER_BASE_URL. Store OPERATOR_API_KEY in your server secret store. Send x-provider-api-key on each request, or Authorization: Bearer, using only one method. Never place keys in URLs, frontend bundles or logs. Operator identity comes from the key; omit operator_id from provider requests. The wallet signing secret is a separate credential.

During onboarding, provide website origins, approved HTTPS return/lobby URLs, wallet callback base URLs, requested currencies and a technical contact. Credentials and configuration are separate for each environment. Your server needs HTTPS/JSON, synchronized clocks, raw-body HMAC verification, durable transaction records and atomic decimal wallet updates. Hosted platforms must permit a custom provider adapter. Static websites need a backend; native apps need a tested browser/WebView integration.

3. Discover assigned games

GET /api/provider/v1/games?provider_id=playngo returns data.items for games assigned and enabled for your operator. GET /api/provider/v1/games/{symbol} returns one game. Read symbols from this catalog and tolerate unknown response fields. Availability does not promise all features, languages or stakes. The v1 contract has no promotional free-spin allocation endpoint.

The prepared symbols are wildframes, honeyrush, moonprincess, wheelofmictlan, secretofdead, legacyofegypt, hugosadventure, trollhunters2 and demon. honeyrush is the original Honey Rush; Honey Rush Black & Yellow is a different title. Never hardcode the catalog count into your integration.

4. Launch the authenticated player

POST /api/provider/v1/launch-url requires Idempotency-Key. The JSON body requires symbol, external_player_id, currency and return_url. Optional lang defaults to en; device is desktop or mobile and defaults to desktop. Use EUR, an assigned symbol and an approved HTTPS return URL in this sandbox. Unsupported game languages are rejected. Derive external_player_id from the authenticated server session; never trust a player ID supplied directly by the browser.

Player IDs are opaque strings of 1–128 characters and preserve case, Unicode and leading/trailing whitespace. Do not trim or normalize them into another identity. Preserve session_id, round_id, transaction_id and request_id exactly. Version 1.0.0 trimmed player IDs; old retries containing whitespace can now return 409. Reconcile the original session before creating a deliberately new launch.

Persist one launch key and the exact body before sending. Identical retries return the original response and original expiry; a different body with the same key returns 409 idempotency_conflict. Retain retry keys for at least 24 hours. After a ticket is consumed or expires, an intentional new launch needs a new key. A timeout is not permission to generate another key.

5. Open and close the game safely

A successful launch returns data.launch_url, session_id and expires_at. Return only the URL to the correct authenticated browser and open it promptly using a redirect or iframe. The ticket is single use and valid for 60 seconds. Use an iframe title, fullscreen permission, no-referrer policy and a responsive height. Your CSP frame-src must allow the assigned game origin; your origin must also be approved for framing by the provider. Test mobile layouts and blocked third-party cookies. CORS does not control iframe embedding.

A session lasts up to 30 minutes idle or four hours absolute. A network reconnect within the same loaded page uses a separate session credential. Reloading the page, restarting the browser or reopening a copied URL needs a fresh launch. Close the previous session with POST /api/provider/v1/sessions/{session_id}/close, then request a new launch with a new key. Closing is repeatable and blocks new play while preserving accepted payouts/refunds. Unused tickets reserve capacity until expiry. Returning to the lobby does not prove settlement is complete.

6. Verify wallet callbacks

Implement POST /balance, /debit, /credit and /rollback beneath your configured wallet base URL. The provider sends content-type: application/json, x-pp-request-id, x-pp-timestamp (Unix seconds) and x-pp-signature. Compute lowercase hexadecimal HMAC-SHA256 over ASCII(timestamp) + a period + the exact raw UTF-8 body, using your wallet signing secret. Compare in constant time and reject timestamp skew greater than 300 seconds. Preserve raw bytes before any JSON middleware; parsing and reserializing changes the signature.

After verifying the signature, parse JSON and require the header request ID to equal body request_id, operator_id to match your configured operator and transaction_type to match the endpoint. Validate player/session ownership and the full callback schema before accessing the ledger. Retries use fresh timestamps, signatures and request IDs while preserving the transaction ID and financial fields. The downloadable deterministic fixture exercises all four language helpers without contacting a wallet.

HMAC-SHA256(wallet_secret, ASCII(timestamp) + "." + raw_request_body)

7. Apply the wallet contract

Amounts and balances are decimal strings with exactly two fractional digits, such as "2.00". Use integer minor units or a decimal type, never binary floating-point arithmetic. Enabled currencies use two fractional digits and no automatic FX conversion. Common callback fields are operator_id, player_id, session_id, game_symbol, currency, request_id, transaction_id, transaction_type, action_type and amount. The OpenAPI schema defines exact field types and limits.

Balance uses amount "0.00" and action_type balance, omitting round fields. Debit requires a positive amount, round_id and round_closed false. Credit requires round_id and round_closed, permits zero, and a final credit closes the round. Rollback requires a positive amount, round_closed true and original_transaction_id. Multiple stages/free spins can share a round; deduplicate by transaction ID, never by round alone. action_type is descriptive and does not replace transaction_type.

Return HTTP 200 with success true, the authoritative balance, matching currency and transaction_id. Valid business rejections use HTTP 200 with success false and an error such as insufficient_funds, unknown_player, currency_mismatch, player_blocked or transaction_cancelled. Invalid signatures or stale timestamps use 401; invalid ownership 403; invalid payload 400; conflicting transaction identity 409; temporary failure 503. Return uncompressed UTF-8 JSON within five seconds, at most 65,536 bytes; never an HTML error page.

{"success":true,"balance":"98.00","currency":"EUR","transaction_id":"txn_debit_example_01"}

8. Guarantee one financial effect

Within each environment/operator, claim a unique transaction_id with a fingerprint of player, session, game, round, type, action, currency, amount, round-close flag and original debit reference. Delivery request IDs and timestamps are excluded. In ONE database transaction: claim/check the ID, lock or atomically update the wallet, validate funds/currency, save the ledger entry and immutable response, then commit. Concurrent duplicates must obtain the same stored result. Persist definitive rejections too; changed financial fields with the same ID are an idempotency_conflict.

A repeated successful financial callback returns its ORIGINAL saved balance and response even if later transactions changed the wallet. Never apply it twice. A separate balance request reads the current wallet. Retain deduplication/cancellation records while retries or reconciliation remain possible; unresolved records must not expire. Retention and archival policy are agreed during onboarding.

9. Handle rollback and recovery

Rollback references the original debit through original_transaction_id. Amount, currency and ownership must match. If the debit exists and is not cancelled, refund once; if already cancelled, succeed without another refund. If the debit is missing, persist a cancellation marker and succeed without changing funds, then reject any later debit with that ID. Serialize against BOTH rollback and original debit IDs so different rollback attempts cannot refund twice. Never reverse an unrelated credit.

An unknown response can mean the wallet committed before the connection failed. Hold play for an unknown debit and resolve the same transaction or cancel it through rollback. Keep credits and refunds pending until confirmed; retries retain their transaction IDs. The provider uses bounded backoff with jitter (approximately 1, 5, 15 then 60 seconds), checked every five seconds, with durable obligations. Keep settling accepted credits/refunds after browser closure or session expiry. A revealed payable round cannot be cancelled instead of paying its win. Closed accepted rounds are auto-collected after two minutes of inactivity; gamble is disabled in this release.

10. Look up sessions and transactions

GET /api/provider/v1/transactions?limit=50 lists your operator’s transactions newest first. Page sizes are 1–100. Send data.next_cursor unchanged as cursor for the next page; cursors are limited to 2,048 characters. GET /api/provider/v1/transactions/{transaction_id} returns operation, amount, currency, round/session IDs, status, attempt count and latest error. GET /api/provider/v1/sessions/{session_id} returns ownership-safe session metadata. URL-encode opaque IDs as individual path segments.

pending, sending and unknown are still resolving: do not create replacement financial transactions. success is wallet-confirmed; rejected is a definitive debit/balance rejection; cancelled means an unresolved debit was cancelled through rollback. A successful debit later refunded retains its success record with cancelled_by_rollback plus a separate successful rollback. Rejected credits/refunds remain unresolved for recovery and operator attention.

11. Handle errors and capacity

Provider errors contain success false, error, message and request_id. For 400/422 correct the input or unsupported currency/language; 401 correct credentials; 403 check operator/game/origin permissions; 404 refresh the catalog or verify the resource; 409 reconcile the conflicting launch key. For 429 honor Retry-After. For 502/503 or network timeouts, back off and retry a launch with its original stored key/body. GET requests may be retried. Preserve sanitized request IDs for support.

The sandbox limit is 60 provider requests per minute per operator, excluding gameplay and outgoing callbacks. Request bodies are limited to 65,536 bytes and ten seconds (413 for size, 408 for timeout). 503 capacity_reached and 503 maintenance include Retry-After: 5; wait and retry the same launch key. Maintenance blocks new wagering while accepted payouts/refunds continue. Four total/two-per-operator sessions are admission settings, not a high-volume capacity claim. A browser verification page is not an API response: report it with the request ID rather than disabling TLS verification or exposing credentials in a browser.

12. Test before client enablement

Run the included offline signature tests first. Then test catalog/authentication, player isolation, launch expiry/reuse, desktop/mobile iframe behavior, all callbacks, simultaneous duplicates, timeout after commit, rollback before debit, insufficient funds, pending credit recovery and reconnect during features. Reconcile accepted debits to final credits or confirmed cancellations. A health check or working demo does not certify a client wallet.

The resumable wallet-conformance.mjs tool requires Node.js 20+, a dedicated idle sandbox player and at least EUR 1.00 in simulated funds. Copy its example config into a private file and run it with a private journal. Its 17 checks have zero net balance change on success. After connection failure, rerun using the SAME config and journal; never delete the journal or replace transaction IDs to retry. Keep secrets, financial journals and live funds out of this public package. The public lobby’s client selector uses simulated player-123 and is not a real login system.

13. Versioning and support

/api/provider/v1 and playngo-callback-v1 identify the contracts; software version 1.0.1 is separate. Additive optional fields and assigned games may change; ignore unknown response fields and refresh the catalog. Breaking field, signing or financial semantics require a separately versioned contract and agreed migration. Keep the accepted OpenAPI files, wallet acceptance journal, assigned origins/currencies/games/limits and browser acceptance results with your release. Sandbox credentials never become production credentials by changing a URL.

For technical support, provide environment, UTC time, game symbol, session/round/transaction IDs, provider request ID and sanitized errors through the agreed onboarding channel. Never share API keys, wallet signing secrets or active launch URLs. Operator alert delivery and production enablement are separate onboarding steps.
